        @startuml
        
        if (Умова) then (Так)
          :Дія 1;<<#palegreen>>
        else (Ні)
          :Дія 2;<<#pink>>
        endif

        @enduml

2. Неповне розгалуження (if-else)

        @startuml

          :Дія 1;<<#palegreen>>
        else (Ні)
         label sp_lab0 
        endif

        @enduml
